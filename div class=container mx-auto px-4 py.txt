            <div class="bg-white p-6 rounded-xl shadow-sm border border-gray-100">
                <h2 class="text-xl font-bold text-emerald-900 mb-3">The Bioscience Scenario</h2>
                <p class="text-sm text-gray-600 leading-relaxed mb-4">
                    In Malaysian intensive broiler production, maximizing feed efficiency is vital due to rising feed costs. This tool models the growth of commercial broilers (e.g., Cobb 500) over a standard 42-day production cycle using a realistic <strong>Gompertz Growth Function</strong>.
                </p>
                <p class="text-sm text-gray-600 leading-relaxed">
                    By applying the <strong>first derivative $f'(x)$</strong>, we pinpoint the exact day of inflection where daily weight gain peaks. Harvesting near or slightly after this peak ensures the best balance between feed consumption and meat yield.

                    <div class="bg-gray-50 p-3 rounded font-mono text-xs overflow-x-auto text-emerald-700">

<script>
    // Mathematical Constants for Gompertz Curve (Cobb 500 estimation adjusted for tropical climate open-house/tunnel parameters)
    const A = 3200;   // Asymptotic maximum weight (g)
    const B = 4.4;    // Initial growth scaling constant
    const K = 0.078;  // Growth rate coefficient

    // Generate data points for 42 days
    const labels = [];
    const weightData = [];
    const rateData = [];

    for (let x = 0; x <= 42; x++) {
        labels.push(x);
        
        // f(x) = A * exp(-B * exp(-K * x))
        const fx = A * Math.exp(-B * Math.exp(-K * x));
        weightData.push(parseFloat(fx.toFixed(1)));
        
        // f'(x) = f(x) * B * K * exp(-K * x)
        const fPrimeX = fx * B * K * Math.exp(-K * x);
        rateData.push(parseFloat(fPrimeX.toFixed(1)));
    }

    // Setup Chart.js
    const ctx = document.getElementById('growthChart').getContext('2d');
    const growthChart = new Chart(ctx, {
        type: 'line',
        data: {
            labels: labels,
            datasets: [
                {
                    label: 'Broiler Weight f(x) [grams]',
                    data: weightData,
                    borderColor: '#065f46',
                    backgroundColor: 'rgba(6, 95, 70, 0.05)',
                    borderWidth: 3,
                    yAxisID: 'yWeight',
                    tension: 0.2,
                    pointRadius: 2,
                    pointHoverRadius: 6
                },
                {
                    label: 'Daily Gain Rate f\'(x) [grams/day]',
                    data: rateData,
                    borderColor: '#b45309',
                    backgroundColor: 'transparent',
                    borderWidth: 2,
                    borderDash: [5, 5],
                    yAxisID: 'yRate',
                    tension: 0.2,
                    pointRadius: 0,
                    pointHoverRadius: 6
                }
            ]
        },
        options: {
            responsive: true,
            maintainAspectRatio: false,
            interaction: {
                mode: 'index',
                intersect: false,
            },
            animation: {
                duration: 2000,
                easing: 'easeOutQuart'
            },
            scales: {
                x: {
                    title: {
                        display: true,
                        text: 'Bird Age (Days)',
                        font: { weight: 'bold' }
                    }
                },
                yWeight: {
                    type: 'linear',
                    display: true,
                    position: 'left',
                    title: {
                        display: true,
                        text: 'Live Weight (g)',
                        font: { weight: 'bold', color: '#065f46' }
                    },
                    min: 0,
                    max: 3500
                },
                yRate: {
                    type: 'linear',
                    display: true,
                    position: 'right',
                    title: {
                        display: true,
                        text: 'Growth Rate (g/day)',
                        font: { weight: 'bold', color: '#b45309' }
                    },
                    grid: {
                        drawOnChartArea: false // only want the grid lines for one axis
                    },
                    min: 0,
                    max: 110
                }
            },
            plugins: {
                legend: {
                    position: 'top',
                },
                tooltip: {
                    callbacks: {
                        footer: function() {
                            return "Click chart to analyze this point.";
                        }
                    }
                }
            },
            onClick: (e) => {
                const points = growthChart.getElementsAtEventForMode(e, 'index', { intersect: false }, true);
                if (points.length > 0) {
                    const index = points[0].index;
                    updateMetrics(index);
                }
            }
        }
    });

    function updateMetrics(day) {
        const weight = weightData[day];
        const rate = rateData[day];

        // Reveal UI panels
        document.getElementById('instruction-text').classList.add('hidden');
        document.getElementById('explanation-text').classList.remove('hidden');
        document.getElementById('metrics-box').classList.remove('hidden');

        // Set numeric metrics
        document.getElementById('metric-day').innerText = `Day ${day}`;
        document.getElementById('metric-weight').innerText = `${weight}g`;
        document.getElementById('metric-rate').innerText = `${rate}g/d`;

        // Layman bioscience dynamic definitions
        let laymanExplanation = "";
        if (day <= 14) {
            laymanExplanation = `<strong>Starter Phase:</strong> The chick is developing vital organs and skeletal frameworks. While the overall weight is low ($f(x) = ${weight}\text{g}$), the growth speed ($f'(x)$) is climbing rapidly every single day.`;
        } else if (day > 14 && day <= 21) {
            laymanExplanation = `<strong>Grower Phase:</strong> Accelerating muscle development. The bird's structural frame is established, and muscle mass accumulation is hitting high velocity.`;
        } else if (day > 21 && day <= 26) {
            laymanExplanation = `<strong>Peak Physiological Acceleration:</strong> At around Day 24, the derivative $f'(x)$ hits its absolute maximum (~92g/day). This is the biological inflection point; the bird will never gain weight faster per day than right now.`;
        } else if (day > 26 && day <= 35) {
            laymanExplanation = `<strong>Finisher Phase:</strong> The flock is still getting heavier, but the *rate* of daily gain ($f'(x)$) is shrinking. Every extra day from here requires significantly more feed to produce the same gram of muscle.`;
        } else {
            laymanExplanation = `<strong>Harvest Threshold:</strong> The derivative has dropped sharply. Biological growth has slowed down drastically. Keeping the birds past this window decreases profit margins due to excessive feed maintenance requirements.`;
        }

        document.getElementById('explanation-text').innerHTML = laymanExplanation;
        
        // Visual accent flash animation on update
        const displayBox = document.getElementById('interactive-display');
        displayBox.classList.remove('from-amber-50', 'to-orange-50');
        displayBox.classList.add('from-emerald-50', 'to-emerald-100');
        setTimeout(() => {
            displayBox.classList.remove('from-emerald-50', 'to-emerald-100');
            displayBox.classList.add('from-amber-50', 'to-orange-50');
        }, 300);
    }
</script>
